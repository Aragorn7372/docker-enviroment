105 public async Task<PageResponseDto<PurchasedResponseDto>> FindAllAsync(FilterPurchasedDto filter)
106 {
107 logger.LogInformation("Obteniendo todos los pedidos con filtro: Página {Page}, Tamaño {Size}", filter.Page, filter.Size);
108 var (paginatedItems, totalCount) = await repository.FindAllAsync(filter);
109 int totalPages = filter.Size > 0 ? (int)Math.Ceiling(totalCount / (double)filter.Size) : 0;
110
111 var allUserIds = paginatedItems
112 .SelectMany(p => new[] { p.PlayerId, p.AssignedTo })
113 .Distinct()
114 .ToList();
115
116 var userDict = new Dictionary<Ulid, UserResponseDto>();
117 var missingIds = new List<Ulid>();
118
119 foreach (var id in allUserIds)
120 {
121 var cached = await cache.GetAsync<UserResponseDto>(CacheKeys.UserDataKey + id);
122 if (cached is not null)
123 userDict[id] = cached;
124 else
125 missingIds.Add(id);
126 }
127
128 if (missingIds.Count != 0)
129 {
130 var dbUsers = await userRepository.FindByIdsAsync(missingIds);
131 foreach (var dbUser in dbUsers)
132 {
133 var dto = dbUser.ToDto(cloudinary);
134 await cache.SetAsync(CacheKeys.UserDataKey + dbUser.Id, dto, TimeSpan.FromMinutes(10));
135 userDict[dbUser.Id] = dto;
136 }
137 }
138
139 var items = new List<PurchasedResponseDto>();
140 foreach (var item in paginatedItems)
141 {
142 if (!userDict.TryGetValue(item.PlayerId, out var playerDto) ||
143 !userDict.TryGetValue(item.AssignedTo, out var encorderDto))
144 {
145 logger.LogWarning("Jugador o encordador no encontrado para el pedido {PurchasedId}", item.Id);
146 continue;
147 }
148 items.Add(item.ToDto(playerDto, encorderDto));
149 }
150
151 logger.LogInformation("Se obtuvieron {ItemCount} pedidos de un total de {TotalCount}", items.Count, totalCount);
152 return new PageResponseDto<PurchasedResponseDto>(
153 Content: items,
154 TotalPages: totalPages,
155 TotalElements: totalCount,
156 PageSize: filter.Size,
157 PageNumber: filter.Page,
158 TotalPageElements: items.Count,
159 SortBy: filter.SortBy,
160 Direction: filter.Direction
161 );
162 }

840 private async Task<Result<Unit, DomainErrors>> UpdateUserWithRetryAsync(
841 User player,
842 int maxRetries = 3)
843 {
844 int attempt = 0;
845
846 while (attempt < maxRetries)
847 {
848 try
849 {
850 await userRepository.UpdateAsync(player);
851 logger.LogInformation("Usuario actualizado exitosamente en intento {Attempt}", attempt + 1);
852 return Result.Success<Unit, DomainErrors>(Unit.Value);
853 }
854 catch (DbUpdateConcurrencyException ex)
855 {
856 attempt++;
857 logger.LogWarning("Conflicto de concurrencia al actualizar usuario. Intento {Attempt} de {MaxRetries}",
858 attempt, maxRetries);
859
860 if (attempt >= maxRetries)
861 {
862 logger.LogError("Fallaron todos los reintentos para actualizar usuario. Error: {Error}", ex.Message);
863 return Result.Failure<Unit, DomainErrors>(
864 new ConcurrencyError("El usuario fue modificado por otra operación después de 3 reintentos. Intente de nuevo."));
865 }
866
867 // Recargar el usuario desde la base de datos para el siguiente intento
868 var reloadedPlayer = await userRepository.FindByIdAsync(player.Id);
869 if (reloadedPlayer is null)
870 {
871 logger.LogError("No se pudo recargar el usuario {PlayerId} para reintento", player.Id);
872 return Result.Failure<Unit, DomainErrors>(
873 new UserNotFoundError("Usuario no encontrado después de conflicto de concurrencia"));
874 }
875
876 // Aplicar los cambios de bonos al usuario recargado
877 reloadedPlayer.Bonos = player.Bonos;
878 player = reloadedPlayer;
879 }
880 }
881
882 return Result.Failure<Unit, DomainErrors>(
883 new ConcurrencyError("No se pudo actualizar el usuario después de varios intentos."));
884 }
