252 {
253 HttpRequestMessage? capturedRequest = null;
254 _mockHandler.Protected()
255 .Setup<Task<HttpResponseMessage>>("SendAsync",
256 ItExpr.IsAny<HttpRequestMessage>(),
257 ItExpr.IsAny<CancellationToken>())
258 .Callback<HttpRequestMessage, CancellationToken>((req, _) => capturedRequest = req)
259 .ReturnsAsync(new HttpResponseMessage(HttpStatusCode.OK));
260
261 var parameters = new Dictionary<string, string> { { "name", "Juan" } };
262 await _service.SendTemplateMessageAsync("+34600000000", "welcome", parameters);
263
264 capturedRequest.Should().NotBeNull();

266 using var doc = JsonDocument.Parse(body);
267 var root = doc.RootElement;
268

270 root.GetProperty("to").GetString().Should().Be("+34600000000");
271 root.GetProperty("type").GetString().Should().Be("template");
272 root.GetProperty("template").GetProperty("name").GetString().Should().Be("welcome");
273 root.GetProperty("template").GetProperty("language").GetProperty("code").GetString().Should().Be("es_ES");
