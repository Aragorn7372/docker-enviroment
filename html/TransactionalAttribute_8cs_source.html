89 public override async Task OnActionExecutionAsync(ActionExecutingContext context, ActionExecutionDelegate next)
90 {
91 var transactions = new List<(IDbContextTransaction Transaction, Type ContextType)>();
92
93 try
94 {
95 // Obtener todas las instancias de DbContext e iniciar transacciones
96 foreach (var contextType in _contextTypes)
97 {
98 var dbContext = context.HttpContext.RequestServices.GetService(contextType) as DbContext;
99 if (dbContext == null)
100 throw new InvalidOperationException($"No se pudo obtener la instancia de {contextType.Name}");
101
102 // Skip transactions for in-memory database (dev mode)
103 if (dbContext.Database.IsInMemory())
104 {
105 Log.Debug("In-memory database detectada, saltando transacción para {DbContextName}", contextType.Name);
106 continue;
107 }
108
109 var transaction = await dbContext.Database.BeginTransactionAsync();
110 transactions.Add((transaction, contextType));
111
112 Log.Information("Transacción iniciada en {DbContextName}", contextType.Name);
113 }
114
115 // Ejecutar la acción
116 var resultContext = await next();
117
118 // Detectar si hay excepción o Result.Failure (HÍBRIDO)
119 bool hasError = resultContext.Exception != null || HasResultFailure(resultContext);
120
121 if (!hasError)
122 {
123 // COMMIT: Si todo está bien
124 foreach (var (transaction, contextType) in transactions)
125 {
126 await transaction.CommitAsync();
127 Log.Information("Transacción confirmada para {DbContextName}", contextType.Name);
128 }
129
130 Log.Information("Todas las transacciones fueron confirmadas exitosamente");
131 }
132 else
133 {
134 // ROLLBACK: Si hay excepción o Result.Failure
135 var reversedTransactions = transactions.AsEnumerable().Reverse();
136 foreach (var (transaction, contextType) in reversedTransactions)
137 {
138 try
139 {
140 await transaction.RollbackAsync();
141 Log.Warning("Transacción revertida para {DbContextName}", contextType.Name);
142 }
143 catch (Exception ex)
144 {
145 Log.Error(ex, "Error durante el rollback de transacción en {DbContextName}", contextType.Name);
146 }
147 }
148
149 if (resultContext.Exception != null)
150 Log.Warning("Transacciones revertidas por excepción: {Exception}",
151 resultContext.Exception?.Message);
152 else
153 Log.Warning("Transacciones revertidas por Result.Failure");
154 }
155 }
156 finally
157 {
158 // Garantizar la liberación de todas las transacciones
159 foreach (var (transaction, contextType) in transactions)
160 {
161 try
162 {
163 await transaction.DisposeAsync();
164 Log.Debug("Transacción liberada para {DbContextName}", contextType.Name);
165 }
166 catch (Exception ex)
167 {
168 Log.Error(ex, "Error liberando transacción para {DbContextName}", contextType.Name);
169 }
170 }
171 }
172 }

190 private bool HasResultFailure(ActionExecutedContext context)
191 {
192 try
193 {
194 if (context.Exception != null)
195 return true;
196
197 if (context.Result is ObjectResult objResult)
198 {
199 if (objResult.StatusCode.HasValue && objResult.StatusCode >= 400)
200 {
201 Log.Debug("Detectado error HTTP {StatusCode} en respuesta", objResult.StatusCode);
202 return true;
203 }
204 }
205
206 if (context.Result is BadRequestResult or NotFoundResult)
207 {
208 Log.Debug("Detectado resultado de error: {ResultType}", context.Result.GetType().Name);
209 return true;
210 }
211
212 if (context.Result is StatusCodeResult statusCodeResult && statusCodeResult.StatusCode >= 400)
213 {
214 Log.Debug("Detectado StatusCodeResult {StatusCode}", statusCodeResult.StatusCode);
215 return true;
216 }
217
218 Log.Debug("Detectado resultado exitoso: {ResultType}", context.Result?.GetType().Name ?? "null");
219 return false;
220 }
221 catch (Exception ex)
222 {
223 Log.Debug(ex, "Error detectando fallo; continuando con commit");
224 return false;
225 }
