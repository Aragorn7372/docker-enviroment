304 private runFilter(q: string): void {
305 const term = q.toLowerCase().trim();
306 if (!term) { this.results.set([]); return; }
307 const matches = this.allData
308 .filter(r => `${r.title} ${r.subtitle} ${r.meta || ''}`.toLowerCase().includes(term))
309 .slice(0, 30);
310 this.results.set(matches);
311 this.selectedIndex.set(0);
312 }

314 groupedResults() {
315 const groups: Record<string, { type: ResultType; items: SearchResult[] }> = {};
316 this.results().forEach(r => {
317 if (!groups[r.type]) groups[r.type] = { type: r.type, items: [] };
318 groups[r.type].items.push(r);
319 });
320 return Object.values(groups);
321 }

323 getGlobalIndex(type: ResultType, localIndex: number): number {
324 let offset = 0;
325 for (const g of this.groupedResults()) {
326 if (g.type === type) return offset + localIndex;
327 offset += g.items.length;
328 }
329 return 0;
330 }

356 @HostListener('window:keydown', ['$event'])
357 onKeydown(event: KeyboardEvent) {
358 if (!this.isOpenSig()) return;
359 const results = this.results();
360 if (event.key === 'Escape') {
361 this.close();
362 event.preventDefault();
363 return;
364 }
365 if (results.length === 0) return;
366 if (event.key === 'ArrowDown') {
367 event.preventDefault();
368 this.selectedIndex.update(i => (i + 1) % results.length);
369 } else if (event.key === 'ArrowUp') {
370 event.preventDefault();
371 this.selectedIndex.update(i => (i - 1 + results.length) % results.length);
372 } else if (event.key === 'Enter') {
373 event.preventDefault();
374 const selected = results[this.selectedIndex()];
375 if (selected) this.selectItem(selected);
376 }
377 }

384 }
