157 private loadFromStorage(): void {
158 try {
159 const storedUser = localStorage.getItem('atp_user');
160 const storedToken = localStorage.getItem('atp_token');
161 if (storedUser && storedToken) {
162 this.state = {
163 user: JSON.parse(storedUser),
164 token: storedToken
165 };
166 }
167 } catch {
168 this.logout();
169 }
170 }
171
172 private saveToStorage(): void {
173 if (this.state.user) {
174 localStorage.setItem('atp_user', JSON.stringify(this.state.user));
175 } else {
176 localStorage.removeItem('atp_user');
177 }
178 if (this.state.token) {
179 localStorage.setItem('atp_token', this.state.token);
180 } else {
181 localStorage.removeItem('atp_token');
182 }
183 }

185 login(username: string, password: string): Observable<{ success: boolean; user?: AppUser; error?: string }> {
186 return this.http.post<AuthResponseDto>(`${API_URL}/Auth/signin`, { Username: username, Password: password }).pipe(
187 map((response) => {
188 const user = this.mapAuthUser(response.user);
189 this.state = { user, token: response.token };
190 this.saveToStorage();
191 return { success: true, user };
192 }),
193 catchError((error: HttpErrorResponse) => {
194 const message = error.error?.message || 'Credenciales incorrectas';
195 return of({ success: false, error: message });
196 })
197 );
198 }
199
200 signup(username: string, email: string, password: string): Observable<{ success: boolean; user?: AppUser; error?: string }> {
201 return this.http.post<AuthResponseDto>(`${API_URL}/Auth/signup`, { Username: username, Email: email, Password: password }).pipe(
202 map((response) => {
203 const user = this.mapAuthUser(response.user);
204 this.state = { user, token: response.token };
205 this.saveToStorage();
206 return { success: true, user };
207 }),
208 catchError((error: HttpErrorResponse) => {
209 const message = error.error?.message || 'No se pudo completar el registro';
210 return of({ success: false, error: message });
211 })
212 );
213 }

318 private mapBackendRole(role: string): UserRole {
319 switch ((role || '').toUpperCase()) {
320 case 'ADMIN': return 'admin';
321 case 'ENCORDER': return 'encordador';
322 case 'OWNER': return 'owner';
323 case 'SUPERVISOR': return 'supervisor';
324 case 'USER':
325 default: return 'tenista';
326 }
327 }

329 private mapAuthUser(user: AuthUserDto): AppUser {

335 phoneNumber: '',
336 imageUrl: '',
337 name: user.username,
338 role: mappedRole,
339 backendRole: user.role,
340 createdAt: user.createdAt
