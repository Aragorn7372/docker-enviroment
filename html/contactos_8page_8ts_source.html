375 private splitName(fullName: string): { name: string; lastName: string } {
376 const parts = (fullName || '').trim().split(' ');
377 if (parts.length <= 1) {
378 return { name: fullName, lastName: '' };
379 }
380 return {
381 name: parts[0],
382 lastName: parts.slice(1).join(' ')
