198 private mapPurchased(dto: PurchasedResponseDto): PedidoCard {

250 private aggregateStatus(lineas: PedidoLineaResponseDto[] | undefined): string {
251 if (!lineas || lineas.length === 0) return 'PENDING';
252 const statuses = lineas.map((l) => (l.status || '').toUpperCase());
253 if (statuses.every((s) => s === 'CANCELED')) return 'CANCELED';
254 if (statuses.every((s) => s === 'DELIVERED_TOPLAYER')) return 'DELIVERED_TOPLAYER';
255 if (statuses.every((s) => s === 'COMPLETED' || s === 'DELIVERED_TOPLAYER')) return 'COMPLETED';
256 if (statuses.some((s) => s === 'IN_PROGRESS')) return 'IN_PROGRESS';
257 return statuses[0] || 'PENDING';
258 }
