287 public async Task ImportDataAsync(ExportDataDto data)
288 {
289 logger.LogInformation("Importing data in correct order");
290
291 if (data.Tournaments.Any())
292 {
293 var tournamentAssignments = new List<(Ulid TournamentId, List<WorkerMachineAssignment> Assignments)>();
294
295 foreach (var tournament in data.Tournaments)
296 {
297 tournamentAssignments.Add((tournament.Id, tournament.WorkerMachineAssignments.ToList()));
298 tournament.WorkerMachineAssignments = new List<WorkerMachineAssignment>();
299 await talleresDbContext.Partidos.AddAsync(tournament);
300 }
301 await talleresDbContext.SaveChangesAsync();
302 logger.LogInformation("Imported {Count} tournaments", data.Tournaments.Count);
303
304 long nextAssignmentId = 1;
305 foreach (var (tournamentId, assignments) in tournamentAssignments)
306 {
307 var existingTournament = await talleresDbContext.Partidos.FirstOrDefaultAsync(t => t.Id == tournamentId);
308 if (existingTournament != null)
309 {
310 foreach (var assignment in assignments)
311 {
312 existingTournament.WorkerMachineAssignments.Add(new WorkerMachineAssignment
313 {
314 Id = nextAssignmentId++,
315 WorkerId = assignment.WorkerId,
316 MachineName = assignment.MachineName
317 });
318 }
319 }
320 }
321 await talleresDbContext.SaveChangesAsync();
322 logger.LogInformation("Imported worker machine assignments for {Count} tournaments", tournamentAssignments.Count);
323 }
324
325 if (data.Users.Any())
326 {
327 await userDbContext.Users.AddRangeAsync(data.Users);
328 await userDbContext.SaveChangesAsync();
329 logger.LogInformation("Imported {Count} users", data.Users.Count);
330 }
