89 async drainQueue(): Promise<void> {
90 if (this._syncing()) return;
91 if (!this.online()) return;
92 this._syncing.set(true);
93 try {
94 // Process jobs sequentially in creation order
95 // eslint-disable-next-line no-constant-condition
96 while (true) {
97 const jobs = await offlineDB.syncQueue.orderBy('createdAt').limit(1).toArray();
98 if (jobs.length === 0) break;
99 const job = jobs[0];
100 const ok = await this.runJob(job);
101 if (!ok) break;
102 }
103 } finally {
104 this._syncing.set(false);
105 }

108 private async runJob(job: SyncJob): Promise<boolean> {
109 try {
110 const opts: any = { observe: 'body' };
111 if (job.headers) {
112 const filtered: Record<string, string> = {};
113 for (const [k, v] of Object.entries(job.headers)) {
114 // Removemos Content-Type cuando vamos a enviar FormData (el browser lo setea con boundary)
115 if (k.toLowerCase() === 'content-type' && this.isFormDataBody(job.body)) continue;
116 filtered[k] = v;
117 }
118 opts.headers = filtered;
119 }
120 // Reconstruimos FormData si el body fue serializado como tal en el offline interceptor
121 const body = this.rehydrateBody(job.body);
122 let resp: any;
123 switch (job.method) {
124 case 'POST':
125 resp = await firstValueFrom(this.http.post(job.url, body, opts));
126 break;
127 case 'PUT':
128 resp = await firstValueFrom(this.http.put(job.url, body, opts));
129 break;
130 case 'PATCH':
131 resp = await firstValueFrom(this.http.patch(job.url, body, opts));
132 break;
133 case 'DELETE':
134 resp = await firstValueFrom(this.http.delete(job.url, opts));
135 break;
136 }
137
138 // If this was a temp create, replace tempId in cache with real id
139 if (job.tempId && resp && typeof resp === 'object') {
140 await this.replaceTempInCache(job.resource, job.tempId, resp);
141 }
142
143 await offlineDB.syncQueue.delete(job.id!);
144 return true;
145 } catch (err: any) {
146 const status: number | undefined = err?.status;
147 const isNetwork = !status || status === 0;
148 const attempts = (job.attempts ?? 0) + 1;
149
150 if (isNetwork) {
151 this._network.set(navigator.onLine);
152 await offlineDB.syncQueue.update(job.id!, { attempts, lastError: 'network' });
153 return false;
154 }
155
156 // Server rejected the change. Skip the job so the queue does not block.
157 await offlineDB.syncQueue.update(job.id!, { attempts, lastError: `HTTP ${status ?? '?'}` });
158 if (attempts >= 5) {
159 await offlineDB.syncQueue.delete(job.id!);
160 console.warn('[OfflineSync] Job dropped after 5 attempts', job);
161 }
162 return true;
163 }
164 }

166 private async replaceTempInCache(resource: SyncJob['resource'], tempId: string, real: any): Promise<void> {
167 const realId = real?.id ?? real?.Id ?? null;
168 if (!realId) return;
169 const tables = tablesForResource(resource);
170 if (!tables) return;
171
172 const tempKey = String(tempId);
173 const realKey = String(realId);
174
175 await offlineDB.transaction('rw', tables.entity, tables.list, async () => {
176 await tables.entity.delete(tempKey);
177 await tables.entity.put({ key: realKey, data: real, updatedAt: Date.now() });
178
179 const lists = await tables.list.toArray();
180 for (const list of lists) {
181 const items = (list.items as any[]) ?? [];
182 const idx = items.findIndex((it) => String(it?.id ?? it?.Id) === tempKey);
183 if (idx >= 0) {
184 items[idx] = real;
185 await tables.list.put({ ...list, items, updatedAt: Date.now() });
186 }
187 }
188 });
189
190 // Migrar localStorage meta de temp ID a real ID (materiales y cuerdas)
191 try {
192 if (resource === 'materials') {
193 const meta = localStorage.getItem('mat:meta:' + tempKey);
194 if (meta) {
195 localStorage.setItem('mat:meta:' + realKey, meta);
196 localStorage.removeItem('mat:meta:' + tempKey);
197 }
198 } else if (resource === 'cuerdas') {
199 const meta = localStorage.getItem('cuerda:meta:' + tempKey);
200 if (meta) {
201 localStorage.setItem('cuerda:meta:' + realKey, meta);
202 localStorage.removeItem('cuerda:meta:' + tempKey);
203 }
204 }
205 } catch { /* noop */ }
206 }

230 private rehydrateBody(body: unknown): any {
231 if (!this.isFormDataBody(body)) return body;
232 const fields = (body as any).fields as Record<string, any>;
233 const fd = new FormData();
234 for (const [k, v] of Object.entries(fields ?? {})) {
235 if (v === undefined || v === null) continue;
236 fd.append(k, typeof v === 'object' ? JSON.stringify(v) : String(v));
237 }
238 return fd;
239 }
