155 async drainQueue(): Promise<void> {
156 if (this._syncing()) { console.log('[OfflineSync] drainQueue — skipping, already syncing'); return; }
157 if (!this.online()) { console.log('[OfflineSync] drainQueue — skipping, offline (online=', this.online(), '| _network=', this._network(), '| _manualOffline=', this._manualOffline(), ')'); return; }
158 console.log('[OfflineSync] drainQueue — starting, pending jobs:', await offlineDB.syncQueue.count());
159 this._syncing.set(true);
160 let processedAny = false;
161 try {
162 // Process jobs sequentially in creation order
163 // eslint-disable-next-line no-constant-condition
164 while (true) {
165 const jobs = await offlineDB.syncQueue.orderBy('createdAt').limit(1).toArray();
166 if (jobs.length === 0) break;
167 const job = jobs[0];
168 const ok = await this.runJob(job);
169 if (!ok) break;
170 processedAny = true;
171 }
172 } finally {
173 this._syncing.set(false);
174 if (processedAny) {
175 this._refreshKey.update(v => v + 1);
176 }
177 }
178 }

180 private async runJob(job: SyncJob): Promise<boolean> {
181 try {
182 const opts: any = { observe: 'body' };
183 if (job.headers) {
184 const filtered: Record<string, string> = {};
185 for (const [k, v] of Object.entries(job.headers)) {
186 // Removemos Content-Type cuando vamos a enviar FormData (el browser lo setea con boundary)
187 if (k.toLowerCase() === 'content-type' && this.isFormDataBody(job.body)) continue;
188 filtered[k] = v;
189 }
190 opts.headers = filtered;
191 }
192 // Reconstruimos FormData si el body fue serializado como tal en el offline interceptor
193 const body = this.rehydrateBody(job.body);
194 let resp: any;
195 switch (job.method) {
196 case 'POST':
197 resp = await firstValueFrom(this.http.post(job.url, body, opts));
198 break;
199 case 'PUT':
200 resp = await firstValueFrom(this.http.put(job.url, body, opts));
201 break;
202 case 'PATCH':
203 resp = await firstValueFrom(this.http.patch(job.url, body, opts));
204 break;
205 case 'DELETE':
206 resp = await firstValueFrom(this.http.delete(job.url, opts));
207 break;
208 }
209
210 // Update cache with real server response
211 if (resp && typeof resp === 'object') {
212 if (job.tempId) {
213 await this.replaceTempInCache(job.resource, job.tempId, resp);
214 } else {
215 await this.applyCacheFromResponse(job.resource, resp);
216 }
217 }
218
219 await offlineDB.syncQueue.delete(job.id!);
220 return true;
221 } catch (err: any) {
222 const status: number | undefined = err?.status;
223 const isNetwork = !status || status === 0;
224 const attempts = (job.attempts ?? 0) + 1;
225
226 if (isNetwork) {
227 this._network.set(navigator.onLine);
228 await offlineDB.syncQueue.update(job.id!, { attempts, lastError: 'network' });
229 return false;
230 }
231
232 // Server rejected the change. Skip the job so the queue does not block.
233 await offlineDB.syncQueue.update(job.id!, { attempts, lastError: `HTTP ${status ?? '?'}` });
234 if (attempts >= 5) {
235 await offlineDB.syncQueue.delete(job.id!);
236 console.warn('[OfflineSync] Job dropped after 5 attempts', job);
237 }
238 return true;
239 }

242 private async replaceTempInCache(resource: SyncJob['resource'], tempId: string, real: any): Promise<void> {
243 const realId = real?.id ?? real?.Id ?? null;
244 if (!realId) return;
245 const tables = tablesForResource(resource);
246 if (!tables) return;
247
248 const tempKey = String(tempId);
249 const realKey = String(realId);
250
251 await tables.entity.delete(tempKey);
252 await tables.entity.put({ key: realKey, data: real, updatedAt: Date.now() });
253
254 const rplLists = await tables.list.toArray();
255 for (const list of rplLists) {
256 const items = (list.items as any[]) ?? [];
257 const idx = items.findIndex((it) => String(it?.id ?? it?.Id) === tempKey);
258 if (idx >= 0) {
259 items[idx] = real;
260 await tables.list.put({ ...list, items, updatedAt: Date.now() });
261 }
262 }
263
264 // Migrar localStorage meta de temp ID a real ID (materiales y cuerdas)
265 try {
266 if (resource === 'materials') {
267 const meta = localStorage.getItem('mat:meta:' + tempKey);
268 if (meta) {
269 localStorage.setItem('mat:meta:' + realKey, meta);
270 localStorage.removeItem('mat:meta:' + tempKey);
271 }
272 } else if (resource === 'cuerdas') {
273 const meta = localStorage.getItem('cuerda:meta:' + tempKey);
274 if (meta) {
275 localStorage.setItem('cuerda:meta:' + realKey, meta);
276 localStorage.removeItem('cuerda:meta:' + tempKey);
277 }
278 }
279 } catch { /* noop */ }
280 }

283 private async applyCacheFromResponse(resource: SyncJob['resource'], respBody: any): Promise<void> {
284 const tables = tablesForResource(resource);
285 if (!tables) return;
286 const now = Date.now();
287 const id = respBody?.id ?? respBody?.Id ?? null;
288 if (!id) return;
289 await tables.entity.put({ key: String(id), data: respBody, updatedAt: now });
290 const acrLists = await tables.list.toArray();
291 for (const list of acrLists) {
292 const items = (list.items as any[]).map((it) =>
293 String(it?.id ?? it?.Id) === String(id) ? respBody : it
294 );
295 await tables.list.put({ ...list, items, updatedAt: now });
296 }

322 private rehydrateBody(body: unknown): any {
323 if (!this.isFormDataBody(body)) return body;
324 const fields = (body as any).fields as Record<string, any>;
325 const fd = new FormData();
326 for (const [k, v] of Object.entries(fields ?? {})) {
327 if (v === undefined || v === null) continue;
328 fd.append(k, typeof v === 'object' ? JSON.stringify(v) : String(v));
329 }
330 return fd;
331 }
