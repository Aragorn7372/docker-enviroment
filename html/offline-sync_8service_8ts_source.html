101 async drainQueue(): Promise<void> {
102 if (this._syncing()) return;
103 if (!this.online()) return;
104 this._syncing.set(true);
105 try {
106 // Process jobs sequentially in creation order
107 // eslint-disable-next-line no-constant-condition
108 while (true) {
109 const jobs = await offlineDB.syncQueue.orderBy('createdAt').limit(1).toArray();
110 if (jobs.length === 0) break;
111 const job = jobs[0];
112 const ok = await this.runJob(job);
113 if (!ok) break;
114 }
115 } finally {
116 this._syncing.set(false);
117 }
118 }

120 private async runJob(job: SyncJob): Promise<boolean> {
121 try {
122 const opts: any = { observe: 'body' };
123 if (job.headers) {
124 const filtered: Record<string, string> = {};
125 for (const [k, v] of Object.entries(job.headers)) {
126 // Removemos Content-Type cuando vamos a enviar FormData (el browser lo setea con boundary)
127 if (k.toLowerCase() === 'content-type' && this.isFormDataBody(job.body)) continue;
128 filtered[k] = v;
129 }
130 opts.headers = filtered;
131 }
132 // Reconstruimos FormData si el body fue serializado como tal en el offline interceptor
133 const body = this.rehydrateBody(job.body);
134 let resp: any;
135 switch (job.method) {
136 case 'POST':
137 resp = await firstValueFrom(this.http.post(job.url, body, opts));
138 break;
139 case 'PUT':
140 resp = await firstValueFrom(this.http.put(job.url, body, opts));
141 break;
142 case 'PATCH':
143 resp = await firstValueFrom(this.http.patch(job.url, body, opts));
144 break;
145 case 'DELETE':
146 resp = await firstValueFrom(this.http.delete(job.url, opts));
147 break;
148 }
149
150 // If this was a temp create, replace tempId in cache with real id
151 if (job.tempId && resp && typeof resp === 'object') {
152 await this.replaceTempInCache(job.resource, job.tempId, resp);
153 }
154
155 await offlineDB.syncQueue.delete(job.id!);
156 return true;
157 } catch (err: any) {
158 const status: number | undefined = err?.status;
159 const isNetwork = !status || status === 0;
160 const attempts = (job.attempts ?? 0) + 1;
161
162 if (isNetwork) {
163 this._network.set(navigator.onLine);
164 await offlineDB.syncQueue.update(job.id!, { attempts, lastError: 'network' });
165 return false;
166 }
167
168 // Server rejected the change. Skip the job so the queue does not block.
169 await offlineDB.syncQueue.update(job.id!, { attempts, lastError: `HTTP ${status ?? '?'}` });
170 if (attempts >= 5) {
171 await offlineDB.syncQueue.delete(job.id!);
172 console.warn('[OfflineSync] Job dropped after 5 attempts', job);
173 }
174 return true;
175 }
176 }

178 private async replaceTempInCache(resource: SyncJob['resource'], tempId: string, real: any): Promise<void> {
179 const realId = real?.id ?? real?.Id ?? null;
180 if (!realId) return;
181 const tables = tablesForResource(resource);
182 if (!tables) return;
183
184 const tempKey = String(tempId);
185 const realKey = String(realId);
186
187 await offlineDB.transaction('rw', tables.entity, tables.list, async () => {
188 await tables.entity.delete(tempKey);
189 await tables.entity.put({ key: realKey, data: real, updatedAt: Date.now() });
190
191 const lists = await tables.list.toArray();
192 for (const list of lists) {
193 const items = (list.items as any[]) ?? [];
194 const idx = items.findIndex((it) => String(it?.id ?? it?.Id) === tempKey);
195 if (idx >= 0) {
196 items[idx] = real;
197 await tables.list.put({ ...list, items, updatedAt: Date.now() });
198 }
199 }
200 });
201
202 // Migrar localStorage meta de temp ID a real ID (materiales y cuerdas)
203 try {
204 if (resource === 'materials') {
205 const meta = localStorage.getItem('mat:meta:' + tempKey);
206 if (meta) {
207 localStorage.setItem('mat:meta:' + realKey, meta);
208 localStorage.removeItem('mat:meta:' + tempKey);
209 }
210 } else if (resource === 'cuerdas') {
211 const meta = localStorage.getItem('cuerda:meta:' + tempKey);
212 if (meta) {
213 localStorage.setItem('cuerda:meta:' + realKey, meta);
214 localStorage.removeItem('cuerda:meta:' + tempKey);
215 }
216 }
217 } catch { /* noop */ }
218 }

241 /** Convierte el body almacenado en cola al tipo correcto antes de enviarlo. */
242 private rehydrateBody(body: unknown): any {
243 if (!this.isFormDataBody(body)) return body;
244 const fields = (body as any).fields as Record<string, any>;
245 const fd = new FormData();
246 for (const [k, v] of Object.entries(fields ?? {})) {
247 if (v === undefined || v === null) continue;
248 fd.append(k, typeof v === 'object' ? JSON.stringify(v) : String(v));
249 }
250 return fd;
251 }
