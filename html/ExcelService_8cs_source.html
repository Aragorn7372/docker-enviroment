256 public async Task<ExcelImportResultDto> ImportAsync(Ulid userId, Ulid tournamentId, List<string> types, string role, Stream excelStream)
257 {
258 var hasAccess = false;
259
260 if (role == User.UserRoles.ADMIN)
261 {
262 hasAccess = true;
263 }
264 else if (role == User.UserRoles.OWNER)
265 {
266 hasAccess = await excelRepository.IsUserOwnerOfTournamentAsync(userId, tournamentId);
267 }
268
269 if (!hasAccess)
270 {
271 throw new UnauthorizedAccessException("No tienes permisos para importar a este torneo");
272 }
273
274 var validTypes = types.Count > 0
275 ? types.Where(t => ValidTypes.Contains(t.ToLower())).ToList()
276 : ValidTypes;
277
278 logger.LogInformation("Starting import for tournament {TournamentId}, types: {Types}", tournamentId, string.Join(",", validTypes));
279
280 var data = await excelArchiveManager.ReadExcelAsync(excelStream);
281 var result = new ExcelImportResultDto();
282
283 if (validTypes.Contains("users"))
284 {
285 await ImportUsersAsync(data.Users, tournamentId, result);
286 }
287
288 if (validTypes.Contains("materials"))
289 {
290 await ImportMaterialsAsync(data.Materials, tournamentId, result);
291 }
292
293 if (validTypes.Contains("cuerdas"))
294 {
295 await ImportCuerdasAsync(data.Cuerdas, tournamentId, result);
296 }
297
298 if (validTypes.Contains("tournament"))
299 {
300 await ImportTournamentAsync(data.Tournament, userId, result);
301 }
302
303 if (validTypes.Contains("pedidos"))
304 {
305 await ImportPedidosAsync(data.Pedidos, data.PedidoLineas, tournamentId, result);
306 }
307
308 logger.LogInformation("Import completed. Users: {Created} created, {Updated} updated. Materials: {MCreated} created, {MUpdated} updated. Pedidos: {PCreated} created, {PUpdated} updated",
309 result.UsersCreated, result.UsersUpdated, result.MaterialsCreated, result.MaterialsUpdated, result.PedidosCreated, result.PedidosUpdated);
310
311 return result;
