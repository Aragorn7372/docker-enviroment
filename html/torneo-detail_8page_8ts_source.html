75 const email = (this.tenistaForm.email || '').trim();
76 const phone = (this.tenistaForm.phone || '').trim();
77
78 if (!name || name.length < 3) {
79 this.toast.error('Nombre obligatorio', 'Mínimo 3 caracteres');
80 return;
81 }
82 if (email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
83 this.toast.error('Email inválido', 'Introduce un email con formato correcto');
84 return;
85 }

88 const req: ContactoRequestDto = {

90 email: email || undefined,
91 phone: phone || undefined,

95 this.addingTenista = true;

100 this.showAddTenistaModal = false;
101 this.refreshTenistas();

103 error: (err) => {
104 this.addingTenista = false;

107 this.toast.error(

155 private loadExistingPlayers(): void {
156 if (!this.torneo) return;
157 this.loadingExisting = true;
158 // Pedimos solo tenistas (findUsers=true). Excluiremos los que ya están en este torneo.
159 this.usersService.listUsers({ findUsers: true, size: 500, sortBy: 'name', direction: 'asc' }).subscribe({
160 next: (page) => {
161 const tid = this.torneo!.id;
162 const yaEnTorneoIds = new Set(this.tenistasDelTorneo.map(t => t.userId));
163 const candidates = (page.content || [])
164 .filter(u => u.id && !yaEnTorneoIds.has(u.id))
165 .filter(u => (u.tournamentId || '') !== tid);
166 // Mapeamos a UserWithId-like para reutilizar el render
167 this.existingPlayers = candidates.map((u): UserWithId => ({
168 userId: u.id || '',
169 username: u.username,
170 name: u.name,
171 imageUrl: u.imageUrl,
172 email: u.email,
173 role: u.role,
174 tournamentId: u.hasTournament ? '' : undefined
175 }));
176 this.loadingExisting = false;
177 },
178 error: () => { this.loadingExisting = false; this.existingPlayers = []; }
179 });
180 }

212 ngOnInit(): void {
213 const id = this.route.snapshot.paramMap.get('id');
214 if (!id) { this.router.navigate(['/torneos']); return; }
215
216 // Try cache first for offline support
217 const cached = this.cache.getTournamentDetail(id);
218 if (cached) {
219 this.torneo = cached.details;
220 this.workerMachines = cached.workerMachines;
221 this.tournamentUsers = cached.users;
222 this.loading = false;
223 }
224
225 forkJoin([
226 this.tournamentsService.getById(id),
227 this.tournamentsService.getWorkers(id),
228 this.usersService.getTournamentUsers(id).pipe(catchError(() => of([])))
229 ]).subscribe({
230 next: ([details, workers, users]) => {
231 this.torneo = details;
232 this.workerMachines = workers;
233 this.tournamentUsers = users;
234 this.loading = false;
235 // Persist to cache for offline use
236 this.cache.saveTournamentDetail(id, { details, workerMachines: workers, users });
237 },
238 error: () => {
239 if (!cached) {
240 this.error = 'No se pudo cargar el torneo';
241 }
242 this.loading = false;
243 }
244 });
