79export class TournamentPickerComponent implements OnInit {
80 search = '';
81 torneos: TournamentItem[] = [];
82 loading = true;
83
84 @Output() tournamentSelected = new EventEmitter<{ id: string; name: string }>();
85 @Output() closed = new EventEmitter<void>();
86
87 constructor(private tournamentsService: TournamentsService) {}
88
89 ngOnInit(): void {
90 this.load();
91 }
92
93 get filtered(): TournamentItem[] {
94 if (!this.search.trim()) return this.torneos;
95 const q = this.search.toLowerCase();
96 return this.torneos.filter(t => t.name.toLowerCase().includes(q));
97 }
98
99 select(t: TournamentItem): void {
100 this.tournamentSelected.emit({ id: t.id, name: t.name });
101 }
102
103 cancel(): void {
104 this.closed.emit();
105 }
106
107 private load(): void {
108 this.loading = true;
109 this.tournamentsService.list({ page: 0, pageSize: 200 }).subscribe({
110 next: (page) => {
111 this.torneos = page.content;
112 this.loading = false;
113 },
114 error: () => { this.loading = false; }
115 });
116 }
117
118 getStatus(t: TournamentItem): string {
119 const now = new Date();
120 const start = new Date(t.startTournament);
121 const end = new Date(t.endTournament);
122 if (start > now) return 'pendiente';
123 if (end < now) return 'cerrado';
124 return 'activo';
125 }
