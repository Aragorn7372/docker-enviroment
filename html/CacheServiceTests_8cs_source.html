186 .Setup(x => x.GetAsync(It.IsAny<string>(), It.IsAny<CancellationToken>()))

188 _cacheStore.TryGetValue(key, out var val) ? val : null);
189
190 _mockDistributedCache
191 .Setup(x => x.SetAsync(It.IsAny<string>(), It.IsAny<byte[]>(), It.IsAny<DistributedCacheEntryOptions>(), It.IsAny<CancellationToken>()))
192 .Callback<string, byte[], DistributedCacheEntryOptions, CancellationToken>((key, val, _, _) => _cacheStore[key] = val)
193 .Returns(Task.CompletedTask);
194
195 _mockDistributedCache
196 .Setup(x => x.RemoveAsync(It.IsAny<string>(), It.IsAny<CancellationToken>()))
197 .Callback<string, CancellationToken>((key, _) => _cacheStore.Remove(key))

199 }
