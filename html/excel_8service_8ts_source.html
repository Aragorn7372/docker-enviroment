14/**
15 * Centraliza export/import contra el backend.
16 *
17 * Endpoints (todos requieren RequireOwnerRole para advanced; export simple
18 * de torneo requiere RequireSupervisorRole):
19 * - GET /api/Excel/export/{tournamentId} → xlsx supervisor
20 * - GET /api/Excel/advanced/export?tournamentId=X&types=... → xlsx multi-hoja
21 * - POST /api/Excel/advanced/import?tournamentId=X&types=... → ExcelImportResultDto
22 */
23@Injectable({ providedIn: 'root' })
24export class ExcelService {
25 constructor(private http: HttpClient) {}
26
27 /** Export simple del torneo (supervisor o superior). */
28 exportTournament(tournamentId: string): Observable<Blob> {
29 return this.http.get(`${API_URL}/Excel/export/${tournamentId}`, { responseType: 'blob' });
30 }
31
32 /**
33 * Export avanzado de uno o varios tipos del torneo. Si `types` está vacío,
34 * exporta TODOS los disponibles. Solo admin/owner.
35 */
36 exportAdvanced(tournamentId: string, types: ExcelType[] = []): Observable<Blob> {
37 let params = new HttpParams().set('tournamentId', tournamentId);
38 if (types.length > 0) params = params.set('types', types.join(','));
39 return this.http.get(`${API_URL}/Excel/advanced/export`, { params, responseType: 'blob' });
40 }
41
42 /** Atajo: export de un solo tipo. Devuelve un Blob xlsx. */
43 exportType(tournamentId: string, type: ExcelType): Observable<Blob> {
44 return this.exportAdvanced(tournamentId, [type]);
45 }
46
47 /**
48 * Import avanzado: sube un .xlsx con las hojas indicadas en `types`.
49 * Si `types` está vacío, intenta importar TODAS las hojas reconocidas.
50 */
51 importAdvanced(tournamentId: string, types: ExcelType[], file: File): Observable<ExcelImportResultDto> {
52 const formData = new FormData();
53 formData.append('file', file);
54 let params = new HttpParams().set('tournamentId', tournamentId);
55 if (types.length > 0) params = params.set('types', types.join(','));
56 return this.http.post<ExcelImportResultDto>(`${API_URL}/Excel/advanced/import`, formData, { params });
57 }
58
59 /** Atajo: import de un solo tipo. */
60 importType(tournamentId: string, type: ExcelType, file: File): Observable<ExcelImportResultDto> {
61 return this.importAdvanced(tournamentId, [type], file);
62 }

64 /**
65 * Helper para descargar un Blob como archivo en el navegador.
66 * Use con `subscribe(blob => excelService.download(blob, "nombre.xlsx"))`.
67 */
68 download(blob: Blob, filename: string): void {
69 const url = URL.createObjectURL(blob);
70 const a = document.createElement('a');
71 a.href = url;
72 a.download = filename;
73 document.body.appendChild(a);
74 a.click();
75 document.body.removeChild(a);
76 URL.revokeObjectURL(url);
77 }
