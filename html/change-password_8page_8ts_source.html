49 constructor(

55 ngOnInit(): void {
56 this.guid = this.route.snapshot.queryParamMap.get('guid') || '';

58
59 goLogin(): void {
60 this.router.navigate(['/login']);

62
63 submit(): void {
64 if (!this.guid) {
65 this.toast.error('Enlace inválido', 'Falta el token de recuperación');
66 return;
67 }
68 if (this.newPassword.length < 6) {
69 this.toast.error('Contraseña corta', 'Debe tener al menos 6 caracteres');
70 return;
71 }
72 if (this.newPassword !== this.confirmPassword) {
73 this.toast.error('No coinciden', 'Las contraseñas deben coincidir');
74 return;
75 }
76 this.isLoading = true;
77 this.http.post(`${API_URL}/Auth/change-password/${this.guid}`, {
78 newPassword: this.newPassword,
79 confirmPassword: this.confirmPassword
80 }).subscribe({
81 next: () => {
82 this.isLoading = false;
83 this.toast.success('Contraseña actualizada', 'Ya puedes iniciar sesión.');
84 this.router.navigate(['/login']);
85 },
86 error: (err) => {
87 this.isLoading = false;
88 const msg = err?.error?.message || 'No se pudo cambiar la contraseña';
89 this.toast.error('Error', msg);
90 }
91 });
92 }
