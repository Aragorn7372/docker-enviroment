63 private static void UpdateTimestamps(DbContext context)
64 {
65 var now = DateTime.UtcNow;
66
67 foreach (var entry in context.ChangeTracker.Entries<ITimestamped>())
68 {
69 switch (entry.State)
70 {
71 case EntityState.Added:
72 entry.Property(e => e.CreatedAt).CurrentValue = now;
73 entry.Property(e => e.UpdatedAt).CurrentValue = now;
74 break;
75 case EntityState.Modified:
76 entry.Property(e => e.UpdatedAt).CurrentValue = now;
77 break;
78 }
79 }
