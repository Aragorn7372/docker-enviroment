203 string? username = request.Username?.Trim();
204 string? email = request.Email?.Trim();

207 var upload= await cloudinary.UploadWithAutoNameAsync(request.Avatar,id.ToString(),CloudinaryConstants.FOLDER_USUARIOS);
208 user.ImageUrl = upload.ImageUrl;
209 user.CloudinaryPublicId = upload.PublicId;
210 }
211
212 if (request.Name is not null && request.Name.Trim().Length > 0)
213 user.Name = request.Name.Trim();
214
215 if (request.Telefono is not null)
216 user.Phone = request.Telefono.Trim();
217
218 if (!string.IsNullOrEmpty(username))
219 user.Username = username;
220
221 if (!string.IsNullOrEmpty(email))
222 user.Email = email;
223
224 var updated = await repository.UpdateAsync(user);
225 return Result.Success<UserResponseDto, AuthError>(updated.ToDto(cloudinary));
