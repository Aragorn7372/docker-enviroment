81 private async Task<Result<TournamentResponseDetailsDto, DomainErrors>> BuildResponseAsync(Tournaments tournament, User? preloadedOwner = null)
82 {
83 var allIds = tournament.WorkersList
84 .Union(tournament.SupervisorList)
85 .Append(tournament.Owner)
86 .Distinct()
87 .ToList();
88
89 var users = await userRepository.FindByIdsAsync(allIds);
90 var userDict = users.ToDictionary(u => u.Id);
91
92 if (!userDict.ContainsKey(tournament.Owner))
93 {
94 if (preloadedOwner is not null)
95 userDict[preloadedOwner.Id] = preloadedOwner;
96 else
97 {
98 var ownerUser = await userRepository.FindByIdAsync(tournament.Owner);
99 if (ownerUser is null)
100 return Result.Failure<TournamentResponseDetailsDto, DomainErrors>(
101 new UserNotFoundError("owner de torneo no encontrado"));
102 userDict[tournament.Owner] = ownerUser;
103 }
104 }
105
106 var workers = tournament.WorkersList
107 .Where(id => userDict.ContainsKey(id))
108 .Select(id => userDict[id].ToDto(cloudinary))
109 .ToList();
110 var supervisors = tournament.SupervisorList
111 .Where(id => userDict.ContainsKey(id))
112 .Select(id => userDict[id].ToDto(cloudinary))
113 .ToList();
114 var owner = userDict[tournament.Owner].ToDto(cloudinary);
115
116 return Result.Success<TournamentResponseDetailsDto, DomainErrors>(
117 tournament.ToTournamentResponseDetailsDto(workers, owner, supervisors, cloudinary));
118 }
