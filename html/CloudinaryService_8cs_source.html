160 public async Task<string> UploadAsync(IFormFile file, string filename, string folder)

162 if (file is null || file.Length == 0)

164 throw new CloudinaryInvalidParameterException("El archivo es requerido");

166

184
185 var uploadResult = await GetCloudinary().UploadAsync(uploadParams);
186
187 if (uploadResult.Error != null)

194 return uploadResult.PublicId;

343 public string GetImageUrlWithTransformations(string publicId, int? width = null, int? height = null)
344 {
345 if (string.IsNullOrWhiteSpace(publicId))
346 {
347 throw new CloudinaryInvalidParameterException("El public ID es requerido");
348 }
349
350 if (CloudinaryOptions.Current == null)
351 {
352 throw new CloudinaryConfigurationException("CloudinaryOptions no ha sido inicializado");
353 }
354
355 var w = width ?? CloudinaryOptions.Current.Transformations.Width;
356 var h = height ?? CloudinaryOptions.Current.Transformations.Height;
357
358 return GetCloudinary().Api.UrlImgUp
359 .Transform(new Transformation()
360 .Width(w)
361 .Height(h)
362 .Crop(CloudinaryOptions.Current.Transformations.Crop)
363 .Quality(CloudinaryOptions.Current.Transformations.Quality)
364 .FetchFormat("auto"))
365 .ResourceType("image")
366 .Secure(true)
367 .BuildUrl(publicId);
