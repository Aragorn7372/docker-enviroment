181function emptyForUrl(url: string): any {
182 // Si la URL parece un detalle (último segmento sin extensión "?"), devolvemos null. Si parece colección, devolvemos shape paginado vacío.
183 // Heurística simple: lista cuando no termina con un ID conocido.
184 const last = url.split('?')[0].split('/').filter(Boolean).pop() || '';
185 const looksLikeId = /^[0-9a-f]{8,}$/i.test(last) || /^\d+$/.test(last) || /^tmp_/.test(last);
186 if (looksLikeId) return null;
187 return {
188 content: [],
189 totalElements: 0,
190 totalPages: 0,
191 pageSize: 0,
192 pageNumber: 0,
193 totalPageElements: 0,
194 sortBy: 'id',
195 direction: 'asc'
196 };
197}

199function shallowEqual(a: any, b: any): boolean {
200 if (a === b) return true;
201 if (!a || !b || typeof a !== 'object' || typeof b !== 'object') return false;
202 if (isCollectionResponse(a) && isCollectionResponse(b)) {
203 if (a.totalElements !== b.totalElements) return false;
204 if (a.content.length !== b.content.length) return false;
205 return JSON.stringify(a.content) === JSON.stringify(b.content);
206 }
207 return JSON.stringify(a) === JSON.stringify(b);
208}

238 for (const list of applyLists) {
239 const items = [respBody, ...(list.items as any[])];

251async function cacheGetResponse(url: string, resource: ReturnType<typeof detectResource>, body: any): Promise<void> {
252 const tables = tablesForResource(resource);
253 const now = Date.now();
254 if (!tables) {
255 await offlineDB.genericCache.put({ key: listKey(url), data: body, updatedAt: now });
256 return;
257 }
258 if (isCollectionResponse(body)) {
259 await tables.list.put({ key: listKey(url), items: body.content, total: body.totalElements, updatedAt: now });
260 for (const item of body.content) {
261 const id = extractId(item);
262 if (id) await tables.entity.put({ key: id, data: item, updatedAt: now });
263 }
264 } else {
265 const id = extractId(body);
266 if (id) await tables.entity.put({ key: id, data: body, updatedAt: now });
267 else await offlineDB.genericCache.put({ key: listKey(url), data: body, updatedAt: now });

302async function serializeBody(body: any): Promise<{ data: any; isFormData: boolean; droppedFiles: string[] }> {
303 if (body instanceof FormData) {
304 const obj: Record<string, any> = {};
305 const droppedFiles: string[] = [];
306 body.forEach((value: any, key: string) => {
307 // FormDataEntryValue es File | string. Detectamos File por su 'name' + 'size'.
308 const isFile = typeof value === 'object' && value !== null && typeof (value as any).size === 'number' && typeof (value as any).name === 'string';
309 if (isFile) {
310 // Archivos no se pueden encolar offline (no son serializables a IndexedDB de forma fiable).
311 droppedFiles.push(key);
312 } else {
313 obj[key] = value;
314 }
315 });
316 return { data: obj, isFormData: true, droppedFiles };
317 }
318 return { data: body, isFormData: false, droppedFiles: [] };
319}

321async function handleOfflineMutation(
322 method: 'POST' | 'PUT' | 'PATCH' | 'DELETE',
323 url: string,
324 resource: ReturnType<typeof detectResource>,
325 body: any,
326 headers: Record<string, string>,
327 sync: OfflineSyncService
328): Promise<any> {
329 const tables = tablesForResource(resource);
330 const now = Date.now();
331 let tempId: string | undefined;
332
333 // FormData no es serializable; lo convertimos a objeto plano + flag.
334 const { data: serializedBody, isFormData, droppedFiles } = await serializeBody(body);
335 // Aplanamos el body para la optimistic UI (no incluimos archivos)
336 const optimisticInput = isFormData ? normalizeFormDataObject(serializedBody) : serializedBody;
337 let optimistic: any = optimisticInput;
338
339 if (tables) {
340 if (method === 'POST') {
341 tempId = genTempId();
342 optimistic = {
343 ...(optimisticInput as object),
344 id: tempId,
345 _offlineTemp: true,
346 _offlineDroppedFiles: droppedFiles.length ? droppedFiles : undefined,
347 createdAt: new Date(now).toISOString(),
348 updatedAt: new Date(now).toISOString()
349 };
350 await tables.entity.put({ key: tempId!, data: optimistic, updatedAt: now });
351 const postLists = await tables.list.toArray();
352 for (const list of postLists) {
353 const items = [optimistic, ...(list.items as any[])];
354 await tables.list.put({ ...list, items, total: (list.total ?? items.length - 1) + 1, updatedAt: now });
355 }
356 } else if (method === 'PUT' || method === 'PATCH') {
357 const putMatch = url.match(/\/([^/?]+)(?:\?|$)/);
358 const putId = putMatch ? decodeURIComponent(putMatch[1]) : null;
359 if (putId) {
360 const prev = await tables.entity.get(putId);
361 const merged = { ...(prev?.data as object ?? {}), ...(optimisticInput as object), id: putId, updatedAt: new Date(now).toISOString(), _offlinePending: true };
362 optimistic = merged;
363 await tables.entity.put({ key: putId, data: merged, updatedAt: now });
364 const putLists = await tables.list.toArray();
365 for (const list of putLists) {
366 const items = (list.items as any[]).map((it) => (String(it?.id ?? it?.Id) === putId ? merged : it));
367 await tables.list.put({ ...list, items, updatedAt: now });
368 }
369 }
370 } else if (method === 'DELETE') {
371 const delMatch = url.match(/\/([^/?]+)(?:\?|$)/);
372 const delId = delMatch ? decodeURIComponent(delMatch[1]) : null;
373 if (delId) {
374 await tables.entity.delete(delId);
375 const delLists = await tables.list.toArray();
376 for (const list of delLists) {
377 const items = (list.items as any[]).filter((it) => String(it?.id ?? it?.Id) !== delId);
378 await tables.list.put({ ...list, items, total: Math.max((list.total ?? items.length) - 1, items.length), updatedAt: now });
379 }
380 optimistic = null;
381 }
382 }
383 }
384 // Encolamos el job: si era FormData, lo guardamos como objeto plano + flag
385 const jobId = await sync.enqueue({
386 method,
387 url,
388 body: isFormData ? { __formData: true, fields: serializedBody } : serializedBody,
389 headers,
390 resource,
391 tempId
392 });
393 console.log('[OfflineInterceptor] handleOfflineMutation - ENCOLADO id:', jobId, '| url:', url);
394 return optimistic;
395}

397/** Normaliza un objeto que viene de FormData (todos los valores son strings) a tipos JS más útiles. */
398function normalizeFormDataObject(obj: Record<string, any>): Record<string, any> {
399 const out: Record<string, any> = {};
400 for (const key of Object.keys(obj)) {
401 const value = obj[key];
402 const lc = key.toLowerCase();
403 // Heurística: campos numéricos
404 if (typeof value === 'string' && /^(stock|precio|price|calibre|total|amount|cantidad|bonos|tensionV|tensionH|nudos)$/i.test(key)) {
405 const n = Number(value.replace(',', '.'));
406 out[key] = Number.isFinite(n) ? n : value;
407 } else {
408 out[key] = value;
409 }
410 // Alias lowercase first letter para uso interno del front (e.g. Marca→brand)
411 if (lc === 'marca') out['brand'] = value;
412 if (lc === 'modelo') out['model'] = value;
413 if (lc === 'precio') out['price'] = Number((value ?? '0').toString().replace(',', '.')) || 0;
414 if (lc === 'stock') out['stock'] = Number(value) || 0;
415 if (lc === 'calibre') out['calibre'] = Number((value ?? '0').toString().replace(',', '.')) || 0;
416 if (lc === 'type') out['category'] = value;
417 if (lc === 'stringformat') out['stringFormat'] = value;
418 if (lc === 'stringstype') out['stringsType'] = value;
419 }
420 return out;
