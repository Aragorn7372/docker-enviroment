69 get filteredOrders(): ReportOrder[] {
70 const term = this.playerFilter.trim().toLowerCase();
71 const from = this.fromDate ? new Date(this.fromDate).getTime() : null;
72 const to = this.toDate ? new Date(this.toDate).getTime() + 24 * 3600 * 1000 : null;
73 return this.orders.filter((o) => {
74 if (!this.statusFilters[o.status]) return false;
75 if (from && o.createdAtMs < from) return false;
76 if (to && o.createdAtMs > to) return false;
77 if (term) {
78 const haystack = `${o.player} ${o.encorder} ${o.id} ${o.racket}`.toLowerCase();
79 if (!haystack.includes(term)) return false;
80 }

378 private recomputeCharts(): void {
379 const orders = this.filteredOrders;
380
381 // 1) Distribución por estado (donut)
382 const proc = orders.filter(o => o.status === 'proceso').length;
383 const lis = orders.filter(o => o.status === 'lista').length;
384 const ent = orders.filter(o => o.status === 'entregada').length;
385 const anu = orders.filter(o => o.status === 'anulada').length;
386 this.seriesStatus = [proc, lis, ent, anu];
387
388 // 2) Evolución diaria por estado (barras apiladas)
389 const buckets = this.bucketByDay(orders);
390 const days = Object.keys(buckets).sort();
391 const categoriesShort = days.map(d => this.shortDay(d));
392 this.chartOptionsType = {
393 ...this.chartOptionsType,
394 xaxis: { ...this.chartOptionsType.xaxis, categories: categoriesShort }
395 };
396 this.seriesType = [
397 { name: 'En proceso', data: days.map(d => buckets[d].proceso) },
398 { name: 'Listas', data: days.map(d => buckets[d].lista) },
399 { name: 'Entregadas', data: days.map(d => buckets[d].entregada) },
400 { name: 'Anuladas', data: days.map(d => buckets[d].anulada) }
401 ];
402
403 // 3) Facturación diaria (área) — separa pagado vs pendiente
404 const paidByDay = days.map(d => buckets[d].paid);
405 const pendingByDay = days.map(d => buckets[d].pending);
406 this.chartOptionsRevenue = {
407 ...this.chartOptionsRevenue,
408 xaxis: { ...this.chartOptionsRevenue.xaxis, categories: categoriesShort }
409 };
410 this.seriesRevenue = [
411 { name: 'Pagado', data: paidByDay },
412 { name: 'Pendiente', data: pendingByDay }
413 ];
414
415 // 4) Top encordadores (barras horizontales) — top 8
416 const byEncorder = new Map<string, number>();
417 for (const o of orders) {
418 const k = o.encorder || '—';
419 byEncorder.set(k, (byEncorder.get(k) || 0) + 1);
420 }
421 const top = Array.from(byEncorder.entries())
422 .filter(([k]) => k && k !== '—')
423 .sort((a, b) => b[1] - a[1])
424 .slice(0, 8);
425 this.chartOptionsEncorders = {
426 ...this.chartOptionsEncorders,
427 xaxis: { ...this.chartOptionsEncorders.xaxis, categories: top.map(([k]) => k) }
428 };
429 this.seriesEncorders = [{ name: 'Pedidos', data: top.map(([, v]) => v) }];
430 }

432 private bucketByDay(orders: ReportOrder[]): Record<string, {
433 proceso: number; lista: number; entregada: number; anulada: number;
434 paid: number; pending: number;
435 }> {
436 const m: Record<string, any> = {};
437 for (const o of orders) {
438 const day = this.toDayKey(o.createdAtMs);
439 if (!m[day]) m[day] = { proceso: 0, lista: 0, entregada: 0, anulada: 0, paid: 0, pending: 0 };
440 m[day][o.status] += 1;
441 if ((o.paymentStatus || '').toUpperCase() === 'PAID') m[day].paid += o.total;
442 else m[day].pending += o.total;
443 }
444 return m;
445 }

447 private toDayKey(ms: number): string {
448 if (!ms) return '0000-00-00';
449 const d = new Date(ms);
450 const y = d.getFullYear();
451 const mo = String(d.getMonth() + 1).padStart(2, '0');
452 const da = String(d.getDate()).padStart(2, '0');
453 return `${y}-${mo}-${da}`;
454 }

456 private shortDay(iso: string): string {
457 // 'YYYY-MM-DD' → 'DD MMM'
458 if (!iso || iso.length < 10) return iso;
459 const d = new Date(iso);
460 if (isNaN(d.getTime())) return iso;
461 return d.toLocaleDateString('es-ES', { day: '2-digit', month: 'short' });
462 }
