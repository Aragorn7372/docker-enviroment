258 get filteredOrders(): ReportOrder[] {
259 const term = this.playerFilter.trim().toLowerCase();
260 return this.orders.filter((order) => {

262 return false;

265 if (!term) {
266 return true;

269 const haystack = `${order.player} ${order.id}`.toLowerCase();
270 return haystack.includes(term);
